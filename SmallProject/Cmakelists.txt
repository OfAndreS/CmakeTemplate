# Versão mínima recomendada do CMake
cmake_minimum_required(VERSION 3.15)

# Definição do projeto (use um placeholder claro)
project(YourProjectName VERSION 1.0)

# Definir o padrão da linguagem C++
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Criar o alvo executável e listar os ficheiros fonte explicitamente
# É uma prática melhor do que usar file(GLOB...)
add_executable(${PROJECT_NAME}
    src/main.cpp
    src/app_utils.cpp
)

# Adicionar o diretório 'include' para que os cabeçalhos sejam encontrados
# PUBLIC: Aplica-se a este alvo e a qualquer outro que o utilize como dependência
target_include_directories(${PROJECT_NAME} PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
)

# Definir uma macro para aceder facilmente à pasta de recursos no código
# Útil para carregar ficheiros, imagens, etc.
target_compile_definitions(${PROJECT_NAME} PUBLIC
    RESOURCES_PATH="${CMAKE_CURRENT_SOURCE_DIR}/resources/"
)

# --- Opcional: Configuração para MSVC (Windows) ---
# Garante que o runtime do Visual Studio é ligado estaticamente,
# evitando a necessidade de instalar redistribuíveis em outras máquinas.
if(MSVC)
    set_property(TARGET ${PROJECT_NAME} PROPERTY
        MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>"
    )
endif()
